cmake_minimum_required(VERSION 3.20)
project(ToolManagerLauncher LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Static runtime: the launcher must start on a clean PC without any VC++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(launcher_core STATIC src/launcher_core.cpp)
target_include_directories(launcher_core PUBLIC src)
if(MSVC)
  target_compile_options(launcher_core PUBLIC /utf-8 /W4 /permissive-)
else()
  target_compile_options(launcher_core PUBLIC -Wall -Wextra)
endif()

if(WIN32)
  # TM_ICON: the .ico shown for ToolManager.exe in Explorer (scripts/make_icon.py writes it).
  set(TM_ICON "" CACHE FILEPATH "icon for ToolManager.exe")
  if(TM_ICON)
    file(TO_CMAKE_PATH "${TM_ICON}" tm_icon_path)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" "1 ICON \"${tm_icon_path}\"\n")
    add_executable(ToolManager WIN32 src/main_win.cpp "${CMAKE_CURRENT_BINARY_DIR}/icon.rc")
  else()
    add_executable(ToolManager WIN32 src/main_win.cpp)
  endif()
  target_link_libraries(ToolManager PRIVATE launcher_core)
  target_compile_definitions(ToolManager PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MINGW)
    target_link_options(ToolManager PRIVATE -municode -static -static-libgcc -static-libstdc++ -s)
  endif()
endif()

enable_testing()
add_executable(launcher_tests tests/test_launcher_core.cpp)
target_link_libraries(launcher_tests PRIVATE launcher_core)
if(MINGW)
  target_link_options(launcher_tests PRIVATE -static)
endif()
add_test(NAME launcher_tests COMMAND launcher_tests)
